cmake_minimum_required(VERSION 3.16)
project(bamboo-native LANGUAGES CXX)
include(CTest)
find_package(ICU REQUIRED COMPONENTS uc)
add_library(bamboo-native STATIC core.cpp data.cpp adapter.cpp)
set_target_properties(bamboo-native PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(bamboo-native PUBLIC cxx_std_17)
target_include_directories(bamboo-native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(bamboo-native PUBLIC ICU::uc)
option(BAMBOO_BUILD_BENCHMARKS "Build the held-key performance benchmark" OFF)
if(BAMBOO_BUILD_BENCHMARKS)
    add_executable(benchmark-bamboo-repeat tests/benchmark-repeat.cpp)
    target_link_libraries(benchmark-bamboo-repeat PRIVATE bamboo-native)
endif()
if(BUILD_TESTING)
    add_executable(test-bamboo-native tests/test-native.cpp)
    target_link_libraries(test-bamboo-native PRIVATE bamboo-native)
    target_compile_definitions(test-bamboo-native PRIVATE
        BAMBOO_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
    add_test(NAME bamboo-native COMMAND test-bamboo-native)
endif()
